Synopsis

#include <atmi.h>

long tptypes(char *ptr, char *type, char *subtype);

Description

tptypes() reports the type, subtype, and length of a tpalloc'd buffer. type and subtype receive NUL-terminated names (pass NULL to skip either); the return value is the buffer length in the type's units, or -1 with tperrno set on error.

Arguments

ArgumentDescription
ptrA buffer from tpalloc().
type, subtypeOutput buffers of at least 8 and 16 characters, or NULL.

Errors (tperrno)

CodeDescription
TPEINVALptr is not a typed buffer.

Example

Full source: docs/api/examples/buffers.c (build and run with make -C docs/api/examples check TUXDIR=<install>); the lines documenting tptypes() are highlighted.

// tpalloc()/tptypes()/tpfree() example: typed buffer lifecycle.

#include <stdio.h>
#include <string.h>
#include <atmi.h>

int main(void) {
  char *buf = tpalloc("STRING", NULL, 64);
  if (buf == NULL) {
    fprintf(stderr, "tpalloc failed: tperrno=%d\n", tperrno);
    return 1;
  }

  char type[8] = {0};
  char subtype[16] = {0};
  long len = tptypes(buf, type, subtype);
  if (len == -1) {
    fprintf(stderr, "tptypes failed: tperrno=%d\n", tperrno);
    tpfree(buf);
    return 1;
  }
  printf("type=%s subtype=%s len=%ld\n", type, subtype, len);

  strcpy(buf, "hello");

  // tpexport renders a typed buffer into a portable form that
  // tpimport rebuilds
  char exported[256];
  long elen = sizeof(exported);
  if (tpexport(buf, 0, exported, &elen, 0) == -1) {
    fprintf(stderr, "tpexport failed: tperrno=%d\n", tperrno);
    tpfree(buf);
    return 1;
  }

  // tpimport rebuilds the buffer; *obuf points at the target buffer
  // and *olen carries its size (22.x grows the buffer when the data
  // does not fit)
  char *back = tpalloc("STRING", NULL, 128);
  long blen = 128;
  if (back == NULL || tpimport(exported, elen, &back, &blen, 0) == -1) {
    fprintf(stderr, "tpimport failed: tperrno=%d\n", tperrno);
    tpfree(buf);
    tpfree(back);
    return 1;
  }
  printf("roundtrip: %s\n", back);
  tpfree(back);

  tpfree(buf);
  return 0;
}
