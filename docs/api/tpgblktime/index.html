Synopsis

#include <atmi.h>

long tpgblktime(long flags);

Description

tpgblktime() reads back the blocking timeout set with tpsblktime(), in the unit requested by flags. flags 0 (or a bare unit bit) reads the TPBLK_ALL value; TPBLK_NEXT reads the next-call value. The unit bits are honored on read: TPBLK_SECOND by default, TPBLK_MILLISECOND returns the value in milliseconds.

Arguments

ArgumentDescription
flags0 for the TPBLK_ALL value, TPBLK_NEXT for the next-call value, TPBLK_ALL explicitly; OR a unit (TPBLK_SECOND default, TPBLK_MILLISECOND to get the value in milliseconds).

Errors (tperrno)

CodeDescription
TPEINVALUnsupported flags word.

Example

Full source: docs/api/examples/blktime.c (build and run with make -C docs/api/examples check TUXDIR=<install>); the lines documenting tpgblktime() are highlighted.

// tpsblktime()/tpgblktime() example: set a per-thread blocking
// timeout, read it back, and make a call that is subject to it.

#include <stdio.h>
#include <string.h>
#include <atmi.h>

int main() {
  if (tpinit((TPINIT *)NULL) == -1) {
    perror("tpinit");
    return 1;
  }

  char *buf = tpalloc("STRING", NULL, 32);
  strcpy(buf, "hello");
  long len = 32;

  // flags carry the target (TPBLK_ALL persists on the thread, TPBLK_NEXT
  // is consumed by the next blocking call) and the unit (TPBLK_SECOND;
  // 12.x also accepts TPBLK_MILLISECOND, 22.1 rejects it with TPEINVAL).
  // 0 restores waiting forever
  if (tpsblktime(5, TPBLK_ALL | TPBLK_SECOND) == -1) {
    fprintf(stderr, "tpsblktime failed: %s (%d)\n", tpstrerror(tperrno),
            tperrno);
    return 1;
  }

  // reads the timeout back in the requested unit; flags 0 means the
  // TPBLK_ALL value
  long ms = tpgblktime(TPBLK_MILLISECOND);
  printf("blocking timeout: %ld ms\n", ms);

  if (tpcall((char *)"TOUPPER", buf, 0, &buf, &len, 0) == -1) {
    fprintf(stderr, "tpcall failed: %s (%d)\n", tpstrerror(tperrno),
            tperrno);
    return 1;
  }

  printf("reply: %s\n", buf);
  tpfree(buf);
  tpterm();
  return 0;
}
